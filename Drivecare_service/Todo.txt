const api = process.env.fast2sms_api_key;

let otpstore = {};
// Function to send SMS using Fast2SMS
const sendOtpSms = async (mobileNo, otp) => {
  const options = {
    method: "POST",
    url: "https://www.fast2sms.com/dev/bulkV2",
    headers: {
      authorization: api,
      "Content-Type": "application/json",
    },
    data: JSON.stringify({
      route: "v3",
      sender_id: "TXTIND",
      message: `Your OTP is ${otp}`,
      numbers: mobileNo,
    }),
  };

  try {
    const response = await axios(options);
    return response.data;
  } catch (error) {
    console.log(error);
    throw new Error("Failed to send SMS");
  }
};

const handeluserlogin = async (req, res) => {
  const { mobileno } = req.body;

  try {
    if (!mobileno) {
      return res.status(203).json({ msg: "Please enter vaild moibleNo" });
    }
    const user = await users.findOne({ mobileno });
    if (!user) {
      return res.status(404).json({ msg: "User not found" });
    }
    const otp = Math.floor(100000 + Math.random() * 900000);

    otpstore[mobileno] = otp;

    // Send the OTP via SMS using Fast2SMS
    const result = await sendOtpSms(mobileno, otp);
    console.log(`OTP sent successfully: ${result}`);
    res.status(200).json({ success: true, message: "OTP sent successfully!" });
  } catch (error) {
    console.log(error);
    res.status(500).json({ msg: "Something went wrong failed to send otp" });
  }
};
const handlegetuserbyID = async (req, res) => {
  const id = req.params.id;
  if (!id) {
    return res.status(400).json({ msg: "ID is required" });
  }
  try {
    const user = await users.findById(id);
    if (!user) {
      return res.status(404).json({ msg: "User not found" });
    }
    res.status(200).json(user);
  } catch (error) {
    console.error(error);
    res.status(500).json({ msg: "Internal Server Error" });
  }
};

// Route to verify OTP
const veriftotp = (req, res) => {
  const { mobileNo, otp } = req.body;

  // Check if the provided OTP matches the stored OTP for the mobile number
  if (otpstore[mobileNo] && otpstore[mobileNo] == otp) {
    delete otpstore[mobileNo]; // OTP verified, delete it from the store
    res
      .status(200)
      .json({ success: true, message: "OTP verified successfully!" });
  } else {
    res
      .status(400)
      .json({ success: false, message: "Invalid OTP or mobile number" });
  }
};
